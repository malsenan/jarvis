# NOTE: openwakeword is deliberately NOT in this file. Its Linux dependency
# metadata demands tflite-runtime, which has no Python 3.14 wheels, so pip

# everything it actually needs for the ONNX backend.

scipy==1.18.1           # openwakeword runtime dependency
scikit-learn==1.9.0     # openwakeword runtime dependency
requests==2.32.5        # openwakeword model downloader dependency
tqdm==4.70.0            # openwakeword model downloader dependency
faster-whisper==1.2.1   # speech-to-text (CTranslate2 whisper, CPU int8)
piper-tts==1.7.0        # text-to-speech (local, fast on CPU)
